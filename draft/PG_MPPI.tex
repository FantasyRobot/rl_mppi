\begin{algorithm}[t]
\caption{PG-MPPI (Policy-Guided MPPI) for Safe Manipulator Planning}
\label{alg:pg-mppi}
\begin{algorithmic}[1]
\Require Dynamics $x_{t+1}=f(x_t,u_t)$; horizon $H$; samples $K$; temperature $\lambda$;
noise covariance $\Sigma$; stage cost $\ell(x,u)$; terminal cost $\ell_f(x)$;
learned policy $\pi_\phi(a\mid s)$ (e.g., SAC/CD-SAC actor); control constraint set $\mathcal U$;
safety filter $\mathcal F_{\text{safe}}(\cdot)$ (collision-avoidance / constraint projection).
\Ensure Control $u_t^{\text{safe}}$ applied to the robot at each time step.

\State \textbf{(Offline, once)} Train policy $\pi_\phi$ in simulation (e.g., SAC/CD-SAC with goal reward + CDF-based safety shaping).
\State \textbf{Initialize} nominal control sequence $U \!=\! \{u_0,\dots,u_{H-1}\}$ (zeros or from $\pi_\phi$).

\For{$t=0,1,2,\dots$}
    \State Observe current state $x_t$.

    \Statex \textbf{Policy-guided nominal (warm start)}
    \State Predict a nominal rollout under the learned policy:
    \State \hspace{1em} $\hat x_0 \gets x_t$
    \For{$i=0$ to $H-1$}
        \State $\hat u_i \gets \mathbb E[\pi_\phi(\cdot\mid \hat x_i)]$ \Comment{actor mean action}
        \State $\hat x_{i+1} \gets f(\hat x_i,\hat u_i)$
    \EndFor
    \State Set (or blend) the nominal sequence $U \gets \{\hat u_0,\dots,\hat u_{H-1}\}$.

    \Statex \textbf{MPPI update (encapsulated as a function)}
    \State $U \gets \textsc{MPPIUpdate}(x_t, U; f,\,\ell,\,\ell_f,\,H,\,K,\,\lambda,\,\Sigma,\,\mathcal U)$
    \Statex \Comment{\textsc{MPPIUpdate} performs sampling rollouts, cost-to-go $S_k$, weight normalization, and path-integral update}

    \Statex \textbf{Safety filtering and execution}
    \State $u_t^\star \gets u_0$
    \State $u_t^{\text{safe}} \gets \mathcal F_{\text{safe}}(x_t, u_t^\star)$ \Comment{e.g., collision-free projection/filter}
    \State Apply $u_t^{\text{safe}}$ to the robot, obtain next state $x_{t+1}$.

    \Statex \textbf{Receding horizon shift}
    \State $U \gets \{u_1,u_2,\dots,u_{H-1}, u_{H-1}\}$ \Comment{shift left and pad}
\EndFor
\end{algorithmic}
\end{algorithm}
